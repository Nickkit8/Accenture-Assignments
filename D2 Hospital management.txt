5. db.patients.find(
  { department: "Cardiology" },
  { _id: 0, patientName: 1 }
)

6. db.patients.aggregate([
  {
    $group: {
      _id: "$department",
      totalTreatmentCost: { $sum: "$treatmentCost" }
    }
  }
])

7. db.patients.aggregate([
  {
    $group: {
      _id: "$doctor",
      avgTreatmentCost: { $avg: "$treatmentCost" }
    }
  },
  {
    $sort: { avgTreatmentCost: 1 }
  }
])

8. db.patients.aggregate([
  {
    $group: {
      _id: "$department",
      totalTreatmentCost: { $sum: "$treatmentCost" }
    }
  },
  {
    $match: {
      totalTreatmentCost: { $gt: 50000 }
    }
  }
])

9. db.patients.aggregate([
  {
    $group: {
      _id: "$department",
      patientCount: { $sum: 1 }
    }
  },
  {
    $sort: { patientCount: -1 }
  }
])

10. db.patients.aggregate([
  {
    $group: {
      _id: "$doctor",
      totalCost: { $sum: "$treatmentCost" }

  {
    $sort: { totalCost: -1 }
  },
  {
    $limit: 3
  }
])

11. db.patients.aggregate([
  {
    $match: {
      age: { $gt: 40 }

  {
    $group: {

      patientCount: { $sum: 1 }
    }
  }
])

12. db.patients.aggregate([
  {
    $group: {
      _id: "$department",
      avgCost: { $avg: "$treatmentCost" }
    }
  },
  {
    $project: {
      _id: 0,
      department: "$_id",
      avgCost: 1
    }
  },
  {
    $sort: { avgCost: -1 }
  }
])

13. db.patients.aggregate([
